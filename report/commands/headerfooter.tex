\newcommand{\headerfooter}[3]{
    \fancypagestyle{fancy}{%
      \fancyhf{}%
      #1
      \renewcommand{\headrulewidth}{0pt}
      \fancyfoot[L]{#2}
      \fancyfoot[C]{\thepage}
      \fancyfoot[R]{#3}
    }
    
    \fancypagestyle{plain}{%
      \fancyhf{}%
      #1
      \renewcommand{\headrulewidth}{0pt}
      \fancyfoot[L]{#2}
      \fancyfoot[C]{\thepage}
      \fancyfoot[R]{#3}
    }
}
